import numpy as np, matplotlib.pyplot as plt
from collections import Counter

np.random.seed(42)
d = np.random.rand(100)
tr, ts = d[:50], d[50:]
lab = ["Class1" if x<=0.5 else "Class2" for x in tr]

for k in [1,2,3,4,5,20,30]:
    p=[Counter(lab[i]for i in np.argsort(abs(tr-x))[:k]).most_common(1)[0][0]for x in ts]

    print(f"\nk={k}")
    for i,(v,l) in enumerate(zip(ts,p),51):
        print(f"Points x{i} ({v:.4f}) -> {l}")
plt.figure()
plt.scatter(tr,[0]*50,c=['b'if l=="Class1"else'r'for l in lab])
plt.scatter(ts,[1]*50,c=['b'if x=="Class1"else'r'for x in p],marker='x')
plt.title(f"KNN k={k}"); plt.grid(); plt.show()
